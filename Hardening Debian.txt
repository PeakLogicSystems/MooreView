Here is a complete, production-ready Nginx configuration template. It includes strict SSL/TLS hardening parameters, security headers, rate limiting, and the reverse proxy setup for your Node.js application.Step 1: Core Configuration (/etc/nginx/nginx.conf)This global configuration file sets up secure worker profiles, sets up strict rate limiting to prevent Denial of Service (DoS), and loads the ModSecurity WAF module.nginx# /etc/nginx/nginx.conf

user www-data;
worker_processes auto;
pid /run/nginx.pid;
include /etc/nginx/modules-enabled/*.conf; # Loads ModSecurity if installed via apt

events {
    worker_connections 1024;
    multi_accept on;
}

http {
    ##
    # Basic Settings
    ##
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;
    
    # Hide Nginx version information from attackers
    server_tokens off;

    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    ##
    # Logging Settings
    ##
    access_log /var/log/nginx/access.log;
    error_log /var/log/nginx/error.log warn;

    ##
    # Rate Limiting (Barracuda-style DoS Protection)
    # Allocates 10MB zone (~160,000 IPs), allows 10 requests/sec per IP
    ##
    limit_req_zone $binary_remote_addr zone=node_limit:10m rate=10r/s;
    limit_conn_zone $binary_remote_addr zone=conn_limit:10m;

    ##
    # Gzip Compression
    ##
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 5;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml;

    ##
    # Virtual Host Configs
    ##
    include /etc/nginx/conf.d/*.conf;
    include /etc/nginx/sites-enabled/*;
}
Use code with caution.Step 2: Hardened Site Configuration (/etc/nginx/sites-available/node-app)This file configures your actual website. Replace yourdomain.com with your actual domain name and ensure your SSL certificates are generated (e.g., via Let's Encrypt / Certbot).nginx# /etc/nginx/sites-available/node-app

# 1. HTTP to HTTPS Global Redirect
server {
    listen 80;
    listen [::]:80;
    server_name yourdomain.com ://yourdomain.com;
    
    # Return a permanent redirect to the secure site
    return 301 https://$host$request_uri;
}

# 2. Hardened HTTPS Server Block
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name yourdomain.com ://yourdomain.com;

    # --- Web Application Firewall (WAF) ---
    # Uncomment the two lines below if ModSecurity is installed and configured
    # modsecurity on;
    # modsecurity_rules_file /etc/nginx/modsec/main.conf;

    # --- SSL/TLS Hardening (Modern Profile) ---
    ssl_certificate /etc/letsencrypt/live/://yourdomain.com;
    ssl_certificate_key /etc/letsencrypt/live/://yourdomain.com;
    
    # Drop legacy protocols (TLS 1.0, 1.1) and allow only highly secure variants
    ssl_protocols TLSv1.2 TLSv1.3;
    
    # Optimize cipher suites for speed and cryptographic strength
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;

    # Session resumption optimization (saves CPU overhead)
    ssl_session_timeout 1d;
    ssl_session_cache shared:SSL:10m;
    ssl_session_tickets off;

    # OCSP Stapling (speeds up TLS handshake safely)
    ssl_stapling on;
    ssl_stapling_verify on;
    resolver 8.8.8.8 1.1.1.1 valid=300s;
    resolver_timeout 5s;

    # --- Security Headers (Defense in Depth) ---
    # Enforce HTTPS globally for 1 year (HSTS)
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    # Prevent clickjacking attacks
    add_header X-Frame-Options "DENY" always;
    # Prevent MIME-type sniffing vulnerabilities
    add_header X-Content-Type-Options "nosniff" always;
    # Restrict referrer visibility to protect user privacy
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    # Basic Content Security Policy (Adjust this based on what your application loads)
    add_header Content-Security-Policy "default-src 'self' http: https: data: blob: 'unsafe-inline'; frame-ancestors 'none';" always;

    # --- Node.js Reverse Proxy & Rate Limiting ---
    location / {
        # Enforce the connection limits defined in nginx.conf
        limit_req zone=node_limit burst=20 nodelay;
        limit_conn conn_limit 10;

        # Forward clean traffic to your local Node.js process (Port 3000)
        proxy_pass http://127.0.0.1:3000;
        
        # Standard HTTP/1.1 WebSockets supporting headers
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        
        # Retain original client IP context for the Node app to see
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Timeouts preventing hung Node processes from locking Nginx
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
}
