# MooreVIEW sandbox — single-server install (test.mooreview.io)
# Region: nyc1 recommended (separate from Atlanta production).
# Docs: docs/CLOUD_DEPLOY_DO_PHASE1_ATL.md#nyc1-sandbox-testmooreviewio

=== DO resources (nyc1) ===
  - Droplet: sandbox-mooreview-nyc1 (2–4 GB Debian 12)
  - Managed Mongo: mooreview-sandbox-mongo (separate from prod)
  - DNS: test.mooreview.io → droplet public IP
  - Firewall: 22, 80, 443, 8883

=== Build & upload (Windows) ===
  npm run build:phase1-bundles
  # WinSCP both bundles to /tmp/ on sandbox droplet:
  #   mooreview-cloud-*d.tgz  (SaaS app)
  #   mooreview-archive-*.tgz (archive server binary)

=== 1. SaaS ===
  mkdir -p /home/mooreview
  tar xzf /tmp/mooreview-cloud-*d.tgz -C /home/mooreview --strip-components=1
  sed -i 's/\r$//' /home/mooreview/deploy/cloud/debian/*.sh

  mkdir -p /etc/mooreview
  cp /home/mooreview/deploy/cloud/phase1/droplet-sandbox/saas.env.template /etc/mooreview/saas.env
  nano /etc/mooreview/saas.env

  export MOOREVIEW_SOURCE=/home/mooreview MOOREVIEW_INSTALL_DIR=/home/mooreview
  bash /home/mooreview/deploy/cloud/debian/install-saas.sh

  # Mongo allowlist: sandbox droplet public IP
  sudo -u mooreview bash -lc 'cd /home/mooreview && npm run seed'

=== 2. Archive (same box) ===
  mkdir -p /opt/mooreview-archive
  tar xzf /tmp/mooreview-archive-*.tgz -C /opt/mooreview-archive --strip-components=1

  cp /home/mooreview/deploy/cloud/phase1/droplet-sandbox/archive.env.template /etc/mooreview/archive.env
  nano /etc/mooreview/archive.env
  mkdir -p /var/lib/mooreview/archive

  MOOREVIEW_SOURCE=/opt/mooreview-archive MOOREVIEW_ARCHIVE_DIR=/opt/mooreview-archive \
    bash /opt/mooreview-archive/deploy/cloud/debian/install-archive.sh

  curl -s http://127.0.0.1:8090/health

=== 3. TLS + nginx ===
  MOOREVIEW_DOMAIN=test.mooreview.io bash /home/mooreview/deploy/cloud/debian/install-saas.sh
  # or if already installed:
  bash /home/mooreview/deploy/cloud/debian/fix-sandbox-web.sh
  certbot --nginx -d test.mooreview.io
  curl -sI https://test.mooreview.io/login

=== 4. MQTT ===
  cp /home/mooreview/deploy/cloud/phase1/droplet-saas/mqtt.env.template /etc/mooreview/mqtt.env
  nano /etc/mooreview/mqtt.env   # MOOREVIEW_DOMAIN=test.mooreview.io
  bash /home/mooreview/deploy/cloud/debian/enable-saas-mqtt.sh

=== 5. Archive compact (local) ===
  cp /home/mooreview/deploy/cloud/phase1/droplet-sandbox/archive-compact.env.template \
     /etc/mooreview/archive-compact.env
  nano /etc/mooreview/archive-compact.env   # token matches archive.env
  bash /home/mooreview/deploy/cloud/debian/enable-phase1-archive-compact.sh

  sudo -u mooreview bash -lc 'cd /home/mooreview && ARCHIVE_COMPACT_DRY_RUN=1 node scripts/run-archive-compact.js'

=== Optional: edge runtime 3090 ===
  bash /home/mooreview/deploy/cloud/debian/enable-runtime-3090.sh

=== Verify ===
  https://test.mooreview.io/login  (demo / demo)
  mqtts://test.mooreview.io:8883

Promote bundles to Atlanta production after sandbox passes.
